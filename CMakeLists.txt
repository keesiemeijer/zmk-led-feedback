function(_lf_validate_range option minimum maximum)
  if(NOT DEFINED ${option} OR NOT "${${option}}" MATCHES "^-?[0-9]+$")
    message(FATAL_ERROR "${option} must be an integer from ${minimum} to ${maximum}, got \"${${option}}\"")
  endif()
  if("${${option}}" LESS "${minimum}" OR "${${option}}" GREATER "${maximum}")
    message(FATAL_ERROR "${option} must be from ${minimum} to ${maximum}, got \"${${option}}\"")
  endif()
endfunction()

if(CONFIG_ZMK_LED_FEEDBACK)
  foreach(_lf_option IN ITEMS ZMK_LED_FEEDBACK_MIN_BRIGHTNESS ZMK_LED_FEEDBACK_BRIGHTNESS ZMK_LED_FEEDBACK_MAX_BRIGHTNESS ZMK_LED_FEEDBACK_BRIGHTNESS_STEP)
    _lf_validate_range(CONFIG_${_lf_option} 1 255)
  endforeach()
  if(CONFIG_ZMK_LED_FEEDBACK_MIN_BRIGHTNESS GREATER CONFIG_ZMK_LED_FEEDBACK_BRIGHTNESS OR
     CONFIG_ZMK_LED_FEEDBACK_BRIGHTNESS GREATER CONFIG_ZMK_LED_FEEDBACK_MAX_BRIGHTNESS)
    message(FATAL_ERROR "CONFIG_ZMK_LED_FEEDBACK_BRIGHTNESS must be between CONFIG_ZMK_LED_FEEDBACK_MIN_BRIGHTNESS and CONFIG_ZMK_LED_FEEDBACK_MAX_BRIGHTNESS")
  endif()

  if(CONFIG_ZMK_LED_FEEDBACK_LAYER)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_LAYER_COLORS_PER_LAYER_LED_COUNT -1 65535)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_LAYER_DURATION_MS 1 2147483647)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_LAYER_INACTIVITY_TIMEOUT_MINUTES 0 60)
    if(NOT "${CONFIG_ZMK_LED_FEEDBACK_LAYER_ALWAYS_ON_LAYERS}" STREQUAL "" AND NOT CONFIG_ZMK_LED_FEEDBACK_LAYER_ALWAYS_ON_LAYERS MATCHES "^[0-9]+(,[0-9]+)*$")
      message(FATAL_ERROR "CONFIG_ZMK_LED_FEEDBACK_LAYER_ALWAYS_ON_LAYERS must be comma separated nonnegative layer numbers, got \"${CONFIG_ZMK_LED_FEEDBACK_LAYER_ALWAYS_ON_LAYERS}\"")
    endif()
  endif()

  if(CONFIG_ZMK_LED_FEEDBACK_CONNECTION OR CONFIG_ZMK_LED_FEEDBACK_BATTERY)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_BLINK_INTERVAL_MS 100 1000)
  endif()
  if(CONFIG_ZMK_SPLIT)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_REQUIRED_PERIPHERALS -1 16)
    if(CONFIG_ZMK_LED_FEEDBACK_REQUIRED_PERIPHERALS GREATER CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS)
      message(FATAL_ERROR "CONFIG_ZMK_LED_FEEDBACK_REQUIRED_PERIPHERALS must not exceed CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS")
    endif()
    if(NOT "${CONFIG_ZMK_LED_FEEDBACK_SLOT_ORDER}" STREQUAL "")
      if(NOT CONFIG_ZMK_LED_FEEDBACK_SLOT_ORDER MATCHES "^[1-9][0-9]*(,[1-9][0-9]*)*$")
        message(FATAL_ERROR "CONFIG_ZMK_LED_FEEDBACK_SLOT_ORDER must be a comma separated permutation of device numbers 1 through ${CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS}")
      endif()
      string(REPLACE "," ";" _lf_slot_numbers "${CONFIG_ZMK_LED_FEEDBACK_SLOT_ORDER}")
      list(LENGTH _lf_slot_numbers _lf_slot_count)
      if(NOT _lf_slot_count EQUAL CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS)
        message(FATAL_ERROR "CONFIG_ZMK_LED_FEEDBACK_SLOT_ORDER must have exactly ${CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS} entries")
      endif()
      set(_lf_seen_slots "")
      foreach(_lf_slot IN LISTS _lf_slot_numbers)
        if(_lf_slot GREATER CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS OR _lf_slot IN_LIST _lf_seen_slots)
          message(FATAL_ERROR "CONFIG_ZMK_LED_FEEDBACK_SLOT_ORDER must be a permutation of device numbers 1 through ${CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS}")
        endif()
        list(APPEND _lf_seen_slots "${_lf_slot}")
      endforeach()
    endif()
  endif()

  if(CONFIG_ZMK_LED_FEEDBACK_BATTERY)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_BATTERY_WARNING_LEVEL 0 100)
    if(CONFIG_ZMK_LED_FEEDBACK_BATTERY_WARNING_LEVEL EQUAL 1)
      message(FATAL_ERROR "CONFIG_ZMK_LED_FEEDBACK_BATTERY_WARNING_LEVEL must be 0 or 2-100")
    endif()
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_BATTERY_WARNING_BLINKS 1 10)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_BATTERY_DELAY_MS 0 2147483647)
    _lf_validate_range(CONFIG_ZMK_LED_FEEDBACK_BATTERY_DURATION_MS 1 2147483647)
  endif()
endif()
